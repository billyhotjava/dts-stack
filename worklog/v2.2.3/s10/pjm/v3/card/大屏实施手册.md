# v2 大屏实施手册

> 使用方法: 找到你要配置的大屏 → 找到组件中文标签 → 粘贴对应 SQL → 绑定指定字段
>
> 参数说明: 所有 SQL 支持 5 个全局筛选器
> - `{{dateFrom}}` = 统计周期-开始
> - `{{dateTo}}` = 统计周期-结束
> - `{{deptId}}` = 事业部/科室 (空=全部)
> - `{{projectNo}}` = 项目编号/名称 (空=全部, 模糊匹配)
> - `{{riskLevel}}` = 风险等级 (空=全部)

## 通用 SQL 片段（下文引用编号）

### SQL-A: 进度域 KPI（从 DWS 聚合，支持项目筛选）

```sql
SELECT
    COUNT(DISTINCT s.project_no)                 AS "项目总数",
    COUNT(DISTINCT CASE WHEN s.incomplete_cnt > 0 THEN s.project_no END) AS "进行中项目",
    COUNT(DISTINCT CASE WHEN s.incomplete_cnt > 0
          AND (s.overdue_incomplete_unchanged_non_general_cnt > 0
            OR s.overdue_incomplete_changed_non_general_cnt > 0) THEN s.project_no END) AS "延期项目",
    SUM(s.total_cnt)                             AS "节点总数",
    SUM(s.pending_normal_cnt)                    AS "正常待完成",
    SUM(s.due_cnt)                               AS "已到时间节点",
    SUM(s.outside_completed_cnt)                 AS "周期外完成",
    SUM(s.incomplete_cnt)                        AS "未完成总数",
    SUM(s.on_time_cnt)                           AS "按时完成数",
    SUM(s.overdue_completed_cnt)                 AS "超期完成数",
    SUM(s.completed_total_cnt)                   AS "完成总数",
    CASE WHEN (SUM(s.due_cnt) + SUM(s.outside_completed_cnt)) = 0 THEN 0
         ELSE ROUND(SUM(s.completed_total_cnt)::numeric
                  / (SUM(s.due_cnt) + SUM(s.outside_completed_cnt))::numeric * 100, 2)
    END                                          AS "完成率",
    CASE WHEN (SUM(s.due_cnt) + SUM(s.outside_completed_cnt)) = 0 THEN 0
         ELSE ROUND(SUM(s.on_time_cnt)::numeric
                  / (SUM(s.due_cnt) + SUM(s.outside_completed_cnt))::numeric * 100, 2)
    END                                          AS "按时完成率",
    SUM(s.abnormal_pending_non_general_cnt)      AS "不正常待变更",
    SUM(s.overdue_incomplete_unchanged_non_general_cnt) AS "超期未完成未变更",
    SUM(s.overdue_incomplete_changed_non_general_cnt)   AS "超期未完成已变更",
    SUM(s.overdue_completed_unchanged_non_general_cnt)  AS "超期已完成未变更",
    CASE WHEN SUM(s.due_cnt) = 0 THEN 0
         ELSE ROUND((SUM(s.abnormal_pending_non_general_cnt)
                    + SUM(s.overdue_incomplete_unchanged_non_general_cnt))::numeric
                  / SUM(s.due_cnt)::numeric * 100, 2)
    END                                          AS "异常率",
    CASE WHEN SUM(s.due_cnt) = 0 THEN 0
         ELSE ROUND((SUM(s.overdue_incomplete_unchanged_non_general_cnt)
                    + SUM(s.overdue_incomplete_changed_non_general_cnt))::numeric
                  / SUM(s.due_cnt)::numeric * 100, 2)
    END                                          AS "超期率",
    SUM(s.incomplete_high_risk_cnt)              AS "未完成高风险",
    SUM(s.incomplete_mid_risk_cnt)               AS "未完成中风险",
    SUM(s.incomplete_milestone_cnt)              AS "未完成里程碑",
    SUM(s.incomplete_major_cnt)                  AS "未完成重大",
    SUM(s.incomplete_important_cnt)              AS "未完成重要",
    SUM(s.milestone_on_time_cnt)                 AS "里程碑按时完成",
    SUM(s.milestone_overdue_completed_cnt)       AS "里程碑超期完成",
    SUM(s.milestone_pending_cnt)                 AS "里程碑待完成",
    CASE WHEN (SUM(s.incomplete_milestone_cnt) + SUM(s.milestone_on_time_cnt)
             + SUM(s.milestone_overdue_completed_cnt)) = 0 THEN 0
         ELSE ROUND((SUM(s.milestone_on_time_cnt) + SUM(s.milestone_overdue_completed_cnt))::numeric
                  / (SUM(s.incomplete_milestone_cnt) + SUM(s.milestone_on_time_cnt)
                   + SUM(s.milestone_overdue_completed_cnt))::numeric * 100, 2)
    END                                          AS "里程碑完成率",
    CASE WHEN SUM(s.milestone_total_cnt) = 0 THEN 0
         ELSE ROUND(SUM(s.milestone_on_time_cnt)::numeric
                  / SUM(s.milestone_total_cnt)::numeric * 100, 2)
    END                                          AS "里程碑按时完成率",
    SUM(s.high_risk_cnt)                         AS "高风险节点数",
    SUM(s.mid_risk_cnt)                          AS "中风险节点数",
    SUM(s.milestone_total_cnt)                   AS "里程碑总数",
    SUM(s.major_total_cnt)                       AS "重大节点总数",
    SUM(s.important_total_cnt)                   AS "重要节点总数",
    MAX(d.max_delay)                             AS "最大延期天数"
FROM biz_dws_progress_monthly_v2 s
LEFT JOIN (
    SELECT MAX(delay_days) AS max_delay
    FROM biz_dwd_project_node_v2
    WHERE plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
      AND plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
      AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR project_no ILIKE '%' || {{projectNo}} || '%')
) d ON true
WHERE s.plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
  AND s.plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
  AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR s.project_no ILIKE '%' || {{projectNo}} || '%')
```

### SQL-B: 进度域二次指标（健康分/比率）

```sql
WITH p AS (
    SELECT SUM(due_cnt) AS due, SUM(on_time_cnt) AS on_time,
           SUM(incomplete_cnt) AS incomplete, SUM(incomplete_high_risk_cnt) AS hi_inc,
           SUM(milestone_on_time_cnt) AS ms_ot, SUM(milestone_total_cnt) AS ms_t,
           SUM(abnormal_pending_non_general_cnt) AS abn,
           SUM(overdue_incomplete_unchanged_non_general_cnt) AS ov_unc,
           SUM(overdue_incomplete_changed_non_general_cnt) AS ov_chg,
           SUM(incomplete_milestone_cnt) AS inc_ms
    FROM biz_dws_progress_monthly_v2
    WHERE plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
      AND plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
      AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR project_no ILIKE '%' || {{projectNo}} || '%')
)
SELECT
    CASE WHEN p.due=0 THEN 0 ELSE ROUND(p.on_time::numeric/p.due::numeric*100,2) END AS "按时完成率",
    CASE WHEN p.due=0 THEN 0 ELSE ROUND((p.ov_unc+p.ov_chg)::numeric/p.due::numeric*100,2) END AS "超期率",
    CASE WHEN p.due=0 THEN 0 ELSE ROUND((p.abn+p.ov_unc)::numeric/p.due::numeric*100,2) END AS "异常率",
    CASE WHEN p.ms_t=0 THEN 0 ELSE ROUND(p.ms_ot::numeric/p.ms_t::numeric*100,2) END AS "里程碑按时率",
    CASE WHEN p.incomplete=0 THEN 0 ELSE ROUND(p.hi_inc::numeric/p.incomplete::numeric*100,2) END AS "高风险未完成占比",
    ROUND(
        (CASE WHEN p.due=0 THEN 0 ELSE p.on_time::numeric/p.due::numeric*100 END)*0.4
      + (CASE WHEN p.ms_t=0 THEN 0 ELSE p.ms_ot::numeric/p.ms_t::numeric*100 END)*0.3
      + (100-(CASE WHEN p.due=0 THEN 0 ELSE (p.abn+p.ov_unc)::numeric/p.due::numeric*100 END))*0.3
    ,2) AS "进度健康度",
    p.inc_ms*3 + p.hi_inc*2 + p.ov_unc*1 AS "风险预警指数"
FROM p
```

### SQL-D: 质量域二次指标

```sql
WITH q AS (
    SELECT SUM(new_issue_cnt) AS n, SUM(open_issue_cnt) AS o,
           SUM(zero_completed_cnt) AS z, SUM(no_zero_plan_cnt) AS np,
           SUM(open_cat_design) AS d
    FROM biz_dws_quality_monthly_v2
    WHERE period_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
      AND period_month <= to_char({{dateTo}}::date, 'YYYY-MM')
      AND ({{deptId}} IS NULL OR {{deptId}} = '' OR dept = {{deptId}})
      AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR project_no ILIKE '%' || {{projectNo}} || '%')
)
SELECT
    CASE WHEN q.n=0 THEN 0 ELSE ROUND(q.z::numeric/q.n::numeric*100,2) END AS "归零率",
    CASE WHEN q.n=0 THEN 0 ELSE ROUND(q.o::numeric/q.n::numeric*100,2) END AS "现存率",
    CASE WHEN q.n=0 THEN 0 ELSE ROUND((q.n-q.np)::numeric/q.n::numeric*100,2) END AS "计划提交率",
    CASE WHEN q.o=0 THEN 0 ELSE ROUND(q.d::numeric/q.o::numeric*100,2) END AS "设计类占比",
    ROUND(
        (CASE WHEN q.n=0 THEN 0 ELSE q.z::numeric/q.n::numeric*100 END)*0.5
      + (CASE WHEN q.n=0 THEN 0 ELSE (q.n-q.np)::numeric/q.n::numeric*100 END)*0.3
      + (100-(CASE WHEN q.n=0 THEN 0 ELSE q.o::numeric/q.n::numeric*100 END))*0.2
    ,2) AS "质量健康度"
FROM q
```

### SQL-E: 技术状态域 KPI

```sql
SELECT
    COUNT(DISTINCT s.project_no)             AS "涉及项目",
    SUM(s.total_change_cnt)                  AS "变更总数",
    SUM(s.new_cat_i)                         AS "I类更改",
    SUM(s.new_cat_ii)                        AS "II类更改",
    SUM(s.new_cat_iii)                       AS "III类更改",
    SUM(s.file_submitted_not_reviewed_i_ii)  AS "已提出未评审(I,II)",
    SUM(s.file_reviewed_not_signed_i_ii)     AS "已评审未签署(I,II)",
    SUM(s.file_reviewed_signed_i_ii)         AS "已评审已签署(I,II)",
    SUM(s.file_submitted_not_signed_iii)     AS "已提出未签署(III)",
    SUM(s.file_submitted_signed_iii)         AS "已提出已签署(III)",
    SUM(s.file_submitted_not_reviewed_i_ii)+SUM(s.file_reviewed_not_signed_i_ii)
      +SUM(s.file_submitted_not_signed_iii)  AS "未签署项",
    SUM(s.order_unsigned_cat_i)+SUM(s.order_unsigned_cat_ii)+SUM(s.order_unsigned_cat_iii) AS "变更单未签署",
    SUM(s.order_signed_cnt)                  AS "变更单已签署",
    CASE WHEN (SUM(s.order_signed_cnt)+SUM(s.order_unsigned_cat_i)+SUM(s.order_unsigned_cat_ii)+SUM(s.order_unsigned_cat_iii))=0 THEN 0
         ELSE ROUND(SUM(s.order_signed_cnt)::numeric
                  /(SUM(s.order_signed_cnt)+SUM(s.order_unsigned_cat_i)+SUM(s.order_unsigned_cat_ii)+SUM(s.order_unsigned_cat_iii))::numeric*100,2)
    END                                      AS "签署完成率",
    SUM(s.reform_pending_i_ii)               AS "未落实整改",
    SUM(s.reform_done_i_ii)                  AS "已落实整改",
    SUM(s.reform_na_iii)                     AS "不涉及整改"
FROM biz_dws_tech_state_monthly_v2 s
WHERE s.period_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
  AND s.period_month <= to_char({{dateTo}}::date, 'YYYY-MM')
  AND ({{deptId}} IS NULL OR {{deptId}} = '' OR s.dept = {{deptId}})
```

### 备用: SQL-A3 科室负载版

```sql
SELECT
    s.project_no                              AS "项目编号",
    SUM(s.total_cnt)                          AS "节点总数",
    SUM(s.completed_total_cnt)                AS "已完成",
    SUM(s.incomplete_cnt)                     AS "未完成"
FROM biz_dws_progress_monthly_v2 s
WHERE s.plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
  AND s.plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
  AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR s.project_no ILIKE '%' || {{projectNo}} || '%')
GROUP BY s.project_no
ORDER BY SUM(s.incomplete_cnt) DESC
```
