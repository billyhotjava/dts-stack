# v2 大屏优化方案

> 基于 v2 指标体系对 v1 大屏的全面优化
> P0 = 必须改（无数据/误导），P1 = 应该改（缺失高价值指标），P2 = 建议改（标签/精度）

## 新增通用 SQL

### SQL-K: 甘特图（三级树结构）

```sql
-- Level 1: 项目级
WITH project_level AS (
    SELECT
        'project'::text              AS level,
        d.project_no                 AS id,
        NULL::text                   AS parent_id,
        d.project_no                 AS "名称",
        COUNT(*)                     AS "节点总数",
        SUM(CASE WHEN d.is_completed THEN 1 ELSE 0 END) AS "已完成",
        SUM(CASE WHEN d.is_incomplete THEN 1 ELSE 0 END) AS "未完成",
        SUM(CASE WHEN d.risk_level = '高' THEN 1 ELSE 0 END) AS "高风险",
        MIN(d.plan_start_date)       AS "计划开始",
        MAX(d.plan_date)             AS "计划结束",
        COALESCE(MIN(d.actual_start_date), MIN(d.plan_start_date)) AS "实际开始",
        MAX(d.actual_date)           AS "实际结束",
        CASE WHEN COUNT(*) = 0 THEN 0
             ELSE ROUND(SUM(CASE WHEN d.is_completed THEN 1 ELSE 0 END)::numeric / COUNT(*)::numeric * 100, 2)
        END                          AS "完成率",
        NULL::text                   AS "节点类型",
        NULL::text                   AS "完成情况",
        NULL::text                   AS "风险等级",
        0                            AS "延期天数",
        0                            AS sort_order
    FROM biz_dwd_project_node_v2 d
    WHERE d.plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
      AND d.plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
      AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR d.project_no ILIKE '%' || {{projectNo}} || '%')
      AND ({{deptId}} IS NULL OR {{deptId}} = '' OR d.dept = {{deptId}})
    GROUP BY d.project_no
),
-- Level 2: 分系统级
subsystem_level AS (
    SELECT
        'subsystem'::text            AS level,
        d.project_no || '|' || COALESCE(d.subsystem, '(无)') AS id,
        d.project_no                 AS parent_id,
        COALESCE(d.subsystem, '(无分系统)') AS "名称",
        COUNT(*)                     AS "节点总数",
        SUM(CASE WHEN d.is_completed THEN 1 ELSE 0 END) AS "已完成",
        SUM(CASE WHEN d.is_incomplete THEN 1 ELSE 0 END) AS "未完成",
        SUM(CASE WHEN d.risk_level = '高' THEN 1 ELSE 0 END) AS "高风险",
        MIN(d.plan_start_date)       AS "计划开始",
        MAX(d.plan_date)             AS "计划结束",
        COALESCE(MIN(d.actual_start_date), MIN(d.plan_start_date)) AS "实际开始",
        MAX(d.actual_date)           AS "实际结束",
        CASE WHEN COUNT(*) = 0 THEN 0
             ELSE ROUND(SUM(CASE WHEN d.is_completed THEN 1 ELSE 0 END)::numeric / COUNT(*)::numeric * 100, 2)
        END                          AS "完成率",
        NULL::text                   AS "节点类型",
        NULL::text                   AS "完成情况",
        NULL::text                   AS "风险等级",
        0                            AS "延期天数",
        1                            AS sort_order
    FROM biz_dwd_project_node_v2 d
    WHERE d.plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
      AND d.plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
      AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR d.project_no ILIKE '%' || {{projectNo}} || '%')
      AND ({{deptId}} IS NULL OR {{deptId}} = '' OR d.dept = {{deptId}})
    GROUP BY d.project_no, d.subsystem
),
-- Level 3: 节点级
node_level AS (
    SELECT
        'node'::text                 AS level,
        d.node_id                    AS id,
        d.project_no || '|' || COALESCE(d.subsystem, '(无)') AS parent_id,
        d.node_task                  AS "名称",
        1                            AS "节点总数",
        CASE WHEN d.is_completed THEN 1 ELSE 0 END AS "已完成",
        CASE WHEN d.is_incomplete THEN 1 ELSE 0 END AS "未完成",
        CASE WHEN d.risk_level = '高' THEN 1 ELSE 0 END AS "高风险",
        d.plan_start_date            AS "计划开始",
        d.plan_date                  AS "计划结束",
        COALESCE(d.actual_start_date, d.plan_start_date) AS "实际开始",
        d.actual_date                AS "实际结束",
        CASE WHEN d.is_completed THEN 100.00 ELSE 0.00 END AS "完成率",
        d.node_type                  AS "节点类型",
        d.completion_status          AS "完成情况",
        d.risk_level                 AS "风险等级",
        COALESCE(d.delay_days, 0)    AS "延期天数",
        2                            AS sort_order
    FROM biz_dwd_project_node_v2 d
    WHERE d.plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
      AND d.plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
      AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR d.project_no ILIKE '%' || {{projectNo}} || '%')
      AND ({{deptId}} IS NULL OR {{deptId}} = '' OR d.dept = {{deptId}})
      AND ({{riskLevel}} IS NULL OR {{riskLevel}} = '' OR d.risk_level = {{riskLevel}})
)
SELECT * FROM project_level
UNION ALL
SELECT * FROM subsystem_level
UNION ALL
SELECT * FROM node_level
ORDER BY id, sort_order
```

### SQL-L: 责任科室负载分布

```sql
SELECT
    d.dept                           AS "责任科室",
    COUNT(*)                         AS "节点总数",
    SUM(CASE WHEN d.is_completed THEN 1 ELSE 0 END) AS "已完成",
    SUM(CASE WHEN d.is_incomplete THEN 1 ELSE 0 END) AS "未完成",
    SUM(CASE WHEN d.risk_level = '高' THEN 1 ELSE 0 END) AS "高风险",
    CASE WHEN COUNT(*) = 0 THEN 0
         ELSE ROUND(SUM(CASE WHEN d.is_completed THEN 1 ELSE 0 END)::numeric / COUNT(*)::numeric * 100, 2)
    END                              AS "完成率"
FROM biz_dwd_project_node_v2 d
WHERE d.plan_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
  AND d.plan_month <= to_char({{dateTo}}::date, 'YYYY-MM')
  AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR d.project_no ILIKE '%' || {{projectNo}} || '%')
  AND d.dept IS NOT NULL
GROUP BY d.dept
ORDER BY SUM(CASE WHEN d.is_incomplete THEN 1 ELSE 0 END) DESC
```

### SQL-M: 风险分类分布（按 risk_category 聚合）

```sql
SELECT
    d.risk_category                  AS "风险分类",
    COUNT(*)                         AS "数量",
    SUM(CASE WHEN d.risk_level = '高' THEN 1 ELSE 0 END) AS "高",
    SUM(CASE WHEN d.risk_level = '中' THEN 1 ELSE 0 END) AS "中",
    SUM(CASE WHEN d.risk_level = '低' THEN 1 ELSE 0 END) AS "低"
FROM biz_dwd_risk_info_v2 d
WHERE d.submit_month >= to_char({{dateFrom}}::date, 'YYYY-MM')
  AND d.submit_month <= to_char({{dateTo}}::date, 'YYYY-MM')
  AND ({{deptId}} IS NULL OR {{deptId}} = '' OR d.dept = {{deptId}})
  AND ({{projectNo}} IS NULL OR {{projectNo}} = '' OR d.project_no ILIKE '%' || {{projectNo}} || '%')
  AND d.risk_category IS NOT NULL
GROUP BY d.risk_category
ORDER BY COUNT(*) DESC
```
