## 目标

给 `useScreenVisitTracker` 加 vitest 单测，覆盖三大场景。

## 技术设计

```tsx
// src/analytics/pages/screens/hooks/useScreenVisitTracker.test.ts
import { renderHook } from "@testing-library/react";
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import reportsService from "@/api/services/reportsService";
import { useScreenVisitTracker } from "./useScreenVisitTracker";

vi.mock("@/api/services/reportsService", () => ({
  default: { visit: vi.fn().mockResolvedValue({ ok: true }) },
}));

describe("useScreenVisitTracker", () => {
  beforeEach(() => {
    vi.useFakeTimers();
    sessionStorage.clear();
  });

  afterEach(() => {
    vi.useRealTimers();
    vi.clearAllMocks();
  });

  it("fires reportsService.visit after 3s when enabled", async () => {
    renderHook(() => useScreenVisitTracker({ screenId: 1, title: "S", enabled: true }));
    expect(reportsService.visit).not.toHaveBeenCalled();
    await vi.advanceTimersByTimeAsync(3000);
    expect(reportsService.visit).toHaveBeenCalledOnce();
    expect(reportsService.visit).toHaveBeenCalledWith({
      code: "screen-1",
      title: "S",
      url: "/bi/screens/1/preview",
    });
  });

  it("does not fire when unmounted before 3s", async () => {
    const { unmount } = renderHook(() => useScreenVisitTracker({ screenId: 1, title: "S", enabled: true }));
    await vi.advanceTimersByTimeAsync(1500);
    unmount();
    await vi.advanceTimersByTimeAsync(5000);
    expect(reportsService.visit).not.toHaveBeenCalled();
  });

  it("does not re-fire within 30s for same id", async () => {
    const first = renderHook(() => useScreenVisitTracker({ screenId: 1, title: "S", enabled: true }));
    await vi.advanceTimersByTimeAsync(3000);
    expect(reportsService.visit).toHaveBeenCalledOnce();
    first.unmount();

    renderHook(() => useScreenVisitTracker({ screenId: 1, title: "S", enabled: true }));
    await vi.advanceTimersByTimeAsync(3000);
    expect(reportsService.visit).toHaveBeenCalledOnce();  // still 1
  });

  it("swallows visit errors silently", async () => {
    (reportsService.visit as any).mockRejectedValueOnce(new Error("boom"));
    renderHook(() => useScreenVisitTracker({ screenId: 2, title: "B", enabled: true }));
    await vi.advanceTimersByTimeAsync(3000);
    // 不抛即可
  });
});
```
